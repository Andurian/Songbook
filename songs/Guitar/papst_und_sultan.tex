\beginsong{Papst und Sultan}[
    by={Die Streuner},

]

\begin{lilypond}
\score {
  <<
    \chords {
      \override ChordName.font-shape = #'italic
      s4 g1 c d g g2 d e1:m c2 d g1 g
    }
    \relative c' {
      \key g \major
      \time 4/4
      \sectionLabel \markup { \box { "Intro" } }
      \partial 4 d4 | 
      g4. g,8 8 8 b g | c2 ~ 8 e fis g | a4. g8 fis e d c | b2 ~ 8 d e fis |
      \repeat volta 2 {
        g4 d a' d | b2\trill ~ 8 a g fis | e4 a fis d
      }
      \alternative {
        { g2 ~ 8 d e fis }
        { g1 }
      }
   }
  >>
}
\end{lilypond}
